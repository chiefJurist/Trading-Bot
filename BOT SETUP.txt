ALL THESE SHOULD BE DONE WHILE RUNNING AS AN ADMINISTRATOR


sudo apt update

sudo apt upgrade -y 

sudo apt install -y build-essential python3-dev python3-pip python3-venv

python3 -m venv myenv

source myenv/bin/activate

pip install numpy

pip install pandas

pip install ccxt

git clone https://github.com/ta-lib/ta-lib.git

cd ta-lib

git checkout main

sudo apt install cmake

mkdir build

cd build

cmake ..

make

sudo make install

ls /usr/local/lib | grep ta-lib

ls /usr/local/include/ta-lib

cd ~

python -m pip show numpy

python -m pip show pandas

python -m pip show ccxt

deactivate

sudo apt autoremove -y
(Purpose: Removes packages that were installed as dependencies but are no longer needed.
When to use: After running sudo apt upgrade to clean up the system.
Include in your setup? Optional, but useful for keeping the system tidy.)

sudo apt clean
(Purpose: Clears the local repository of downloaded package files, freeing up space.
Include in your setup? Optional, primarily if disk space is a concern.)
